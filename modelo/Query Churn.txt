WITH MAIN AS (


    SELECT
        SE.USER_ID
        , SE.SUBSCRIPTION_ID
        , DATE(SE.EVENT_TIME) AS NEW_PAID_SUBSCRIPTION_DATE
        
        , COALESCE(DATEDIFF('day', S.TRIAL_START_TIME, S.TRIAL_END_TIME), 0) AS TRIAL_DURATION_DAYS
        , S.TIER AS SUBSCRIPTION_TIER
        , S.SOURCE AS SUBSCRIPTION_SOURCE
        , S.PROVIDER AS SUBSCRIPTION_PROVIDER
        , S.AFFILIATE AS SUBSCRIPTION_AFFILIATE
        
        , U.AGE
        , U.GENDER_IDENTITY
        , U.COUNTRY
        , U.REGION_TYPE_LEVEL_1
        , U.REGION_TYPE_LEVEL_2
        , U.CLASS_FORMAT_PREFERENCE
        , U.CLASS_INTENSITY_PREFERENCE
        , U.CLASS_PER_WEEK_GOAL
        , U.DANCE_LEVEL_PREFERENCE
        , U.FITNESS_GOAL_PREFERENCE
        , U.MOTIVATING_FACTORS
        , U.MUSIC_PREFERENCE
        , U.APP_VERSION
        , U.DEVICE_OPERATING_SYSTEM
        , U.DEVICE_MANUFACTURER
        , U.DEVICE_BRAND
        , U.DEVICE_MODEL
    
        , DATEDIFF('day', UA_AI.FIRST_EVENT_TIME, SE.EVENT_TIME) APP_INSTALL_TO_PAID_SUBSCRIPTION_DAYS
    FROM CONSUMER_APP.ANALYTICS.NEW_SUBSCRIPTIONS_EVENTS SE
    
    LEFT JOIN CONSUMER_APP.ANALYTICS.NEW_SUBSCRIPTIONS S ON SE.SUBSCRIPTION_ID = S.SUBSCRIPTION_ID
    LEFT JOIN CONSUMER_APP.ANALYTICS.NEW_USERS U ON SE.USER_ID = U.USER_ID
    LEFT JOIN CONSUMER_APP.ANALYTICS.NEW_USERS_ACTIVITY UA_AI ON SE.USER_ID = UA_AI.USER_ID AND UA_AI.EVENT_NAME = 'APP_INSTALLED'
    
    WHERE TRUE
    AND SE.EVENT_APP_USER_TYPE = 'Consumer'
    AND U.IS_APP_TESTER = FALSE
    AND SE.EVENT_NAME = 'NEW_PAID_SUBSCRIPTION'

)

, VIDEO_EVENTS AS (

    SELECT
        AE.USER_ID
        , COUNT(CASE WHEN AE.EVENT_NAME = 'VIDEO_STARTED' THEN AE.EVENT_NAME ELSE NULL END) AS NO_VIDEO_STARTED_30D
        , COUNT(CASE WHEN AE.EVENT_NAME = 'VIDEO_WATCHED' THEN AE.EVENT_NAME ELSE NULL END) AS NO_VIDEO_WATCHED_30D
        , AVG(AE.VIDEO_WATCHED_PERCENTAGE) AS AVG_VIDEO_WATCHED_PERCENTAGE_30D
        , AVG(VC.VIDEO_LENGTH_MIN) AS AVG_VIDEO_LENGTH_MIN_30D
    FROM CONSUMER_APP.ANALYTICS.NEW_APP_EVENTS AE
    
    LEFT JOIN CONSUMER_APP.ANALYTICS.NEW_VIDEO_CLASSES VC ON AE.VIDEO_NAME = VC.VIDEO_NAME
    LEFT JOIN MAIN M ON AE.USER_ID = M.USER_ID
    
    WHERE TRUE
    AND M.USER_ID IS NOT NULL
    AND AE.EVENT_NAME IN ('VIDEO_STARTED', 'VIDEO_WATCHED')
    AND DATE(AE.EVENT_TIME) >= M.NEW_PAID_SUBSCRIPTION_DATE
    AND DATE(AE.EVENT_TIME) <= M.NEW_PAID_SUBSCRIPTION_DATE + INTERVAL '30 DAY'
    
    GROUP BY ALL

)

, LAST_SUBSCRIPTION_DATE AS (

    SELECT
        SUBSCRIPTION_ID
        , MAX(DATE(EVENT_TIME)) AS LAST_SUBSCRIPTION_DATE
    FROM CONSUMER_APP.ANALYTICS.NEW_SUBSCRIPTIONS_EVENTS

    WHERE TRUE
    AND EVENT_NAME = 'ACTIVE_PAID_SUBSCRIPTION'

    GROUP BY ALL

)

SELECT
    M.*
    , COALESCE(VE.NO_VIDEO_STARTED_30D, 0) AS NO_VIDEO_STARTED_30D
    , COALESCE(VE.NO_VIDEO_WATCHED_30D, 0) AS NO_VIDEO_WATCHED_30D
    , COALESCE(VE.AVG_VIDEO_WATCHED_PERCENTAGE_30D, 0) AS AVG_VIDEO_WATCHED_PERCENTAGE_30D
    , COALESCE(VE.AVG_VIDEO_LENGTH_MIN_30D, 0) AS AVG_VIDEO_LENGTH_MIN_30D
    , DATEDIFF('day', M.NEW_PAID_SUBSCRIPTION_DATE, LSD.LAST_SUBSCRIPTION_DATE) AS PAID_SUBSCRIPTION_DURATION_DAYS
    , CASE WHEN PAID_SUBSCRIPTION_DURATION_DAYS <= 35 THEN 1 ELSE 0 END AS PAID_SUBSCRIPTION_CHURN_30D
FROM MAIN M

LEFT JOIN VIDEO_EVENTS VE ON M.USER_ID = VE.USER_ID
LEFT JOIN LAST_SUBSCRIPTION_DATE LSD ON M.SUBSCRIPTION_ID = LSD.SUBSCRIPTION_ID

WHERE TRUE
AND M.NEW_PAID_SUBSCRIPTION_DATE < '2025-10-01'

;
